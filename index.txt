    <script>
        document.addEventListener('DOMContentLoaded', () => {
            const bookingModal = document.getElementById('booking-modal');
            const openCalendarBtn = document.getElementById('open-calendar-btn');
            const closeBookingBtn = document.getElementById('close-booking-modal-btn');
            const bookingPayEl = document.getElementById('booking-pay');

            const productConfigs = {
                smallPackage: {
                    key: 'smallPackage',
                    price: document.getElementById('small-package-total'),
                    message: document.getElementById('small-package-status'),
                    mount: document.getElementById('small-package-pay-area')
                },
                booking: {
                    key: 'booking',
                    price: document.getElementById('booking-total'),
                    message: document.getElementById('booking-status'),
                    mount: document.getElementById('booking-pay-area')
                }
            };

            const applePayAvailable = !!(window.ApplePaySession && typeof window.ApplePaySession.canMakePayments === 'function' && window.ApplePaySession.canMakePayments());
            const mountedSignatures = { smallPackage: '', booking: '' };
            let lastState = { smallPackage: null, booking: null };

            function formatTotal(value) {
                return typeof value === 'number' && isFinite(value) ? value.toFixed(2) + ' EUR' : '—';
            }

            function fitPayArea(container, button) {
                const wrap = container.firstElementChild;
                if (!wrap || !button) return;
                const containerWidth = container.clientWidth;
                wrap.style.transform = '';
                wrap.style.transformOrigin = '';
                wrap.style.margin = '';
                wrap.style.marginBottom = '';
                if (containerWidth <= 0) return;
                if (button.width > containerWidth) {
                    const scale = containerWidth / button.width;
                    wrap.style.transform = 'scale(' + scale + ')';
                    wrap.style.transformOrigin = 'top left';
                    wrap.style.marginBottom = -(button.height * (1 - scale)) + 'px';
                } else {
                    wrap.style.margin = '0 auto';
                }
            }

            function mountPayArea(config, product) {
                const button = product.button;
                const viewport = product.viewport;
                const mountTime = Date.now();
                const wrap = document.createElement('div');
                wrap.className = 'pay-wrap';
                wrap.style.width = button.width + 'px';
                wrap.style.height = button.height + 'px';
                wrap.style.overflow = 'hidden';
                wrap.style.position = 'relative';
                wrap.style.borderRadius = '12px';
                const iframe = document.createElement('iframe');
                iframe.setAttribute('src', product.url);
                iframe.setAttribute('allow', 'payment');
                iframe.setAttribute('title', 'Apple Pay');
                iframe.style.position = 'absolute';
                iframe.style.left = -button.x + 'px';
                iframe.style.top = -button.y + 'px';
                iframe.style.width = viewport.width + 'px';
                iframe.style.height = viewport.docHeight + 'px';
                iframe.style.border = '0';
                iframe.addEventListener('load', () => {
                    if (Date.now() - mountTime < 10000) return;
                    fetch('/api/consumed/' + config.key, { method: 'POST' }).catch(() => {});
                });
                wrap.appendChild(iframe);
                config.mount.appendChild(wrap);
                fitPayArea(config.mount, button);
            }

            function applyProductState(config, product) {
                config.price.textContent = formatTotal(product ? product.totalEur : null);
                if (!applePayAvailable) {
                    if (mountedSignatures[config.key] !== '') {
                        mountedSignatures[config.key] = '';
                        config.mount.innerHTML = '';
                    }
                    config.mount.style.display = 'none';
                    config.message.textContent = 'Az Apple Pay csak Apple eszközön érhető el.';
                    config.message.style.display = '';
                    return;
                }
                const live = !!(product && product.status === 'live' && product.url && product.button && product.viewport);
                const signature = live ? product.url + '|' + JSON.stringify(product.button) + '|' + JSON.stringify(product.viewport) : '';
                if (signature !== mountedSignatures[config.key]) {
                    mountedSignatures[config.key] = signature;
                    config.mount.innerHTML = '';
                    config.mount.style.display = live ? '' : 'none';
                    if (live) mountPayArea(config, product);
                } else {
                    config.mount.style.display = live ? '' : 'none';
                }
                if (live) {
                    config.message.style.display = 'none';
                } else {
                    config.message.textContent = 'Fizetés előkészítése…';
                    config.message.style.display = '';
                }
            }

            function applyState(data) {
                if (!data) return;
                lastState = { smallPackage: data.smallPackage || null, booking: data.booking || null };
                applyProductState(productConfigs.smallPackage, lastState.smallPackage);
                applyProductState(productConfigs.booking, lastState.booking);
            }

            async function fetchState() {
                try {
                    const response = await fetch('/api/state', { cache: 'no-store' });
                    if (!response.ok) return;
                    applyState(await response.json());
                } catch (err) {}
            }

            setInterval(fetchState, 2000);
            fetchState();

            window.addEventListener('resize', () => {
                if (lastState.smallPackage && lastState.smallPackage.button) fitPayArea(productConfigs.smallPackage.mount, lastState.smallPackage.button);
                if (lastState.booking && lastState.booking.button) fitPayArea(productConfigs.booking.mount, lastState.booking.button);
            });

            let currentDate = new Date();
            currentDate.setDate(1);
            let selectedDate = null;
            let selectedTime = null;
            const availableTimeSlots = ['10:00', '11:00', '14:00', '15:00', '16:00', '18:00'];
            const prevMonthBtn = document.getElementById('prev-month-btn');
            const nextMonthBtn = document.getElementById('next-month-btn');
            const monthYearEl = document.getElementById('current-month-year');
            const calendarDaysEl = document.getElementById('calendar-days');
            const timeSlotsContainer = document.getElementById('time-slots-container');
            const timeSlotsEl = document.getElementById('time-slots');

            function renderCalendar() {
                const now = new Date();
                now.setHours(0, 0, 0, 0);
                const year = currentDate.getFullYear();
                const month = currentDate.getMonth();
                monthYearEl.textContent = year + '. ' + new Intl.DateTimeFormat('hu-HU', { month: 'long' }).format(currentDate);
                calendarDaysEl.innerHTML = '';
                const firstDayOfMonth = new Date(year, month, 1).getDay();
                const daysInMonth = new Date(year, month + 1, 0).getDate();
                let firstDayIndex = firstDayOfMonth === 0 ? 6 : firstDayOfMonth - 1;
                for (let i = 0; i < firstDayIndex; i++) {
                    const emptyEl = document.createElement('div');
                    emptyEl.className = 'calendar-day empty';
                    calendarDaysEl.appendChild(emptyEl);
                }
                for (let day = 1; day <= daysInMonth; day++) {
                    const dayDate = new Date(year, month, day);
                    const dayEl = document.createElement('div');
                    dayEl.className = 'calendar-day';
                    dayEl.textContent = day;
                    if (dayDate < now) {
                        dayEl.classList.add('disabled');
                    } else {
                        dayEl.addEventListener('click', () => {
                            selectedDate = (selectedDate && selectedDate.toDateString() === dayDate.toDateString()) ? null : dayDate;
                            selectedTime = null;
                            updateSelection();
                        });
                    }
                    if (selectedDate && dayDate.toDateString() === selectedDate.toDateString()) {
                        dayEl.classList.add('selected');
                    }
                    calendarDaysEl.appendChild(dayEl);
                }
                updateSelection();
            }

            function renderTimeSlots() {
                if (selectedDate) {
                    timeSlotsContainer.classList.add('active');
                } else {
                    timeSlotsContainer.classList.remove('active');
                    return;
                }
                timeSlotsEl.innerHTML = '';
                availableTimeSlots.forEach(time => {
                    const slotEl = document.createElement('div');
                    slotEl.className = 'time-slot';
                    slotEl.textContent = time;
                    slotEl.addEventListener('click', () => {
                        selectedTime = (selectedTime === time) ? null : time;
                        updateSelection();
                    });
                    if (selectedTime === time) {
                        slotEl.classList.add('selected');
                    }
                    timeSlotsEl.appendChild(slotEl);
                });
            }

            function updateSelection() {
                document.querySelectorAll('#calendar-days .calendar-day').forEach(el => el.classList.remove('selected'));
                if (selectedDate && selectedDate.getFullYear() === currentDate.getFullYear() && selectedDate.getMonth() === currentDate.getMonth()) {
                    const dayEls = Array.from(calendarDaysEl.children).filter(c => !c.classList.contains('empty'));
                    const el = dayEls[selectedDate.getDate() - 1];
                    if (el) el.classList.add('selected');
                }
                renderTimeSlots();
                document.querySelectorAll('.time-slot').forEach(el => {
                    el.classList.toggle('selected', selectedTime === el.textContent);
                });
                const readyToPay = !!(selectedDate && selectedTime);
                bookingPayEl.hidden = !readyToPay;
                if (readyToPay && lastState.booking && lastState.booking.button) {
                    requestAnimationFrame(() => {
                        if (lastState.booking && lastState.booking.button) {
                            fitPayArea(productConfigs.booking.mount, lastState.booking.button);
                        }
                    });
                }
            }

            function closeModal() {
                bookingModal.classList.remove('active');
            }

            openCalendarBtn.addEventListener('click', () => {
                renderCalendar();
                bookingModal.classList.add('active');
            });
            closeBookingBtn.addEventListener('click', closeModal);
            bookingModal.addEventListener('click', e => { if (e.target === bookingModal) closeModal(); });
            document.addEventListener('keydown', e => {
                if (e.key === 'Escape' && bookingModal.classList.contains('active')) {
                    closeModal();
                }
            });
            prevMonthBtn.addEventListener('click', () => {
                currentDate = new Date(currentDate.getFullYear(), currentDate.getMonth() - 1, 1);
                renderCalendar();
            });
            nextMonthBtn.addEventListener('click', () => {
                currentDate = new Date(currentDate.getFullYear(), currentDate.getMonth() + 1, 1);
                renderCalendar();
            });
        });
    </script>
